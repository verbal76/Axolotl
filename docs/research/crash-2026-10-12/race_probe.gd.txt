extends SceneTree
# Reproduction probe for the Godot 4.7.2 audio bus-details race (ledger row 37).
# mode=writes: 16 looping 2D players, volume_db rewritten every frame. mode=3d: 16 looping 3D players
# moving each frame, no volume writes. Quits after `secs`; a crash prints a backtrace instead.
var players: Array = []
var mode := "writes"
var secs := 240.0
var t := 0.0

func _init() -> void:
	for a in OS.get_cmdline_user_args():
		if a.begins_with("mode="): mode = a.substr(5)
		if a.begins_with("secs="): secs = float(a.substr(5))
	var wav := AudioStreamWAV.new()
	wav.format = AudioStreamWAV.FORMAT_16_BITS
	wav.mix_rate = 22050
	var data := PackedByteArray()
	data.resize(22050 * 2)
	for i in 22050:
		data.encode_s16(i * 2, int(sin(i * 0.05) * 8000))
	wav.data = data
	wav.loop_mode = AudioStreamWAV.LOOP_FORWARD
	wav.loop_end = 22050
	if mode == "3d":
		var cam := Camera3D.new()
		root.add_child(cam)
	for i in 16:
		var p: Node = AudioStreamPlayer3D.new() if mode == "3d" else AudioStreamPlayer.new()
		p.stream = wav
		root.add_child(p)
		p.play()
		players.append(p)
	print("RACE start mode=%s" % mode)

func _process(dt: float) -> bool:
	t += dt
	for p in players:
		if mode == "3d":
			(p as Node3D).position = Vector3(randf_range(-20, 20), randf_range(-5, 5), randf_range(-20, 20))
		else:
			p.volume_db = randf_range(-20.0, 0.0)
	if Time.get_ticks_msec() > secs * 1000.0:
		print("RACE survived %d s mode=%s" % [int(secs), mode])
		return true
	return false
